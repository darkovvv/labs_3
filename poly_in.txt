% тест полиномиального калькулятора
[
  многострочный комментарий
  без вложенности
]
Add(2x^2-x+2, -x^2+3x-1); % сложить заданные многочлены
Mult(x^2+3x-1, 2x+x^3); % умножить два многочлена
Add(4x-8); % прибавить к сумматору
Eval(1); % вычислить значение сумматора в точке 1
Div(x^5, x^2-1); % целая часть от деления
Mod(x^5, x^2-1); % остаток от деления
Diff(3x^3-2x^2+5x-7); % производная
Cmps(x^2+1, 2x+3); % композиция (2x+3)^2 + 1
